Я полностью изучил предоставленные package.json, server.js, public/styles.css,
public/app.js и public/index.html.

Что выявлено при анализе совместимости с server.js

Перед переносом дизайна обнаружены важные особенности вашего backend
(server.js), под которые я адаптировал frontend без изменения server.js:

1.  Обязательный cloudPassword при регистрации: POST /api/auth/register и POST
    /api/auth/login в server.js требуют phone, firstName, password и
    cloudPassword (иначе сервер отвечает 400). Форма регистрации в новом
    интерфейсе адаптирована и включает поля «Фамилия» (необязательно), «Облачный
    пароль (2FA) *» и загрузку аватара.
2.  Валюта ton на сервере вместо C-Coins: в server.js поле пользователя
    называется ton, в /api/admin/grant используется action: 'ton', а в NFT_GIFTS
    поле floor содержит строки вида '6 796 TON'. Чтобы не ломать сервер, запросы
    отправляют action: 'ton' и читают user.ton, а в интерфейсе раздел
    отображается как C-Coins (строки TON в карточках каталога на клиенте
    приводятся к C-Coins; при желании позже можно переименовать в server.js).
3.  Стартовый баланс и каналы: server.js создаёт новых пользователей с stars: 0,
    ton: 0 и создаёт только чат «Избранное» (saved_<id>). В соответствии с вашим
    запретом «не имитировать начисления и не создавать фиктивных чатов» клиент
    отображает реальные данные с сервера.
4.  Отсутствовавший обработчик звонков в старом app.js: старый app.js вызывал
    onWsCall(payload) при получении call:signal по WebSocket, но функция нигде
    не была объявлена (входящий вызов падал с ReferenceError). В новой версии
    onWsCall реализована (приём/отклонение входящего вызова, счётчик разговора,
    завершение). При этом сам server.js пересылает только сигналы
    offer/accept/end без WebRTC-медиапотока.
5.  Отказ от внешних картинок nft.fragment.com: вместо загрузки картинок со
    стороннего домена в подарках и NFT используются встроенные векторные
    SVG-иллюстрации из нашего нового дизайна с привязкой к серверным id/slug/bg.
6.  localStorage только для токена: как и в исходном app.js, в localStorage
    остался исключительно cw_token для Bearer-авторизации. Все аккаунты,
    профили, чаты, сообщения, истории, контакты, подарки, маркет, звонки и
    админка работают через реальные API и WebSocket.

(Напоминаю честно: из-за отсутствия доступа к вашей ФС и Node/браузеру в этой
чат-среде резервные копии, запись на диск и живое тестирование не выполнялись —
ниже приведены команды для локального бэкапа и готовые версии трёх файлов.)

async function api(method, url, body = null) {
  const headers = { 'Content-Type': 'application/json' };
  if (token) headers['Authorization'] = 'Bearer ' + token;
  const r = await fetch(url, { method, headers, body: body ? JSON.stringify(body) : null });
  const d = await r.json().catch(() => ({}));
  if (!r.ok) {
    if (r.status === 401 && token) logout(true);
    throw new Error(d.error || 'Ошибка сервера');
  }
  return d;
}

function uploadAvatarFile(inp, prevId, phId, cb) {
  const f = inp.files[0]; if (!f) return;
  const r = new FileReader();
  r.onload = e => {
    const img = new Image();
    img.onload = () => {
      const c = document.createElement('canvas'), max = 360;
      let w = img.width, h = img.height;
      if (w > h && w > max) { h *= max / w; w = max; } else if (h > max) { w *= max / h; h = max; }
      c.width = w; c.height = h;
      c.getContext('2d').drawImage(img, 0, 0, w, h);
      const url = c.toDataURL('image/jpeg', 0.8);
      if (prevId) { const el = $(prevId); el.src = url; el.classList.remove('hidden'); }
      if (phId) $(phId).classList.add('hidden');
      cb(url);
    };
    img.src = e.target.result;
  };
  r.readAsDataURL(f);
}

function setWsIndicator(online) {

  $('ws-status-side')?.classList.toggle('online', online);

function connectWS() {
  if (ws) ws.close();
  const proto = location.protocol === 'https:' ? 'wss:' : 'ws:';
  ws = new WebSocket(`${proto}//${location.host}?token=${encodeURIComponent(token)}`);
  ws.onopen = () => setWsIndicator(true);
  ws.onclose = () => { setWsIndicator(false); if (token) setTimeout(connectWS, 3000); };
  ws.onmessage = ev => {
    try {
      const { event, payload } = JSON.parse(ev.data);
      if (event === 'message:new') {
        const c = chats.find(x => x.id === payload.chatId);
        if (!c) return refreshAllData();
        if (!c.messages.some(m => m.id === payload.message.id)) c.messages.push(payload.message);
        if (activeChatId === payload.chatId) {
          renderMessages();
          api('POST', `/api/chats/${payload.chatId}/read`).catch(() => {});
        } else if (payload.message.senderId !== currentUser.id) {
          c.unread = (c.unread || 0) + 1;
          if (currentUser.settings?.notifications !== false) {
            showToast(`${c.title}: ${payload.message.text || 'Медиа'}`);
          }
        }
        renderChats();
      } else if (event === 'message:edit') {
        const m = chats.find(x => x.id === payload.chatId)?.messages.find(m => m.id === payload.message.id);
        if (m) { m.text = payload.message.text; m.edited = true; if (activeChatId === payload.chatId) renderMessages(); renderChats(); }
      } else if (event === 'message:delete' && payload.forAll) {
        const c = chats.find(x => x.id === payload.chatId);
        if (c) { c.messages = c.messages.filter(m => m.id !== payload.msgId); if (activeChatId === payload.chatId) renderMessages(); renderChats(); }
      } else if (event === 'message:react') {
        const m = chats.find(x => x.id === payload.chatId)?.messages.find(m => m.id === payload.msgId);
        if (m) { m.reactions = payload.reactions; if (activeChatId === payload.chatId) renderMessages(); }
      } else if (event === 'message:read') {
        const c = chats.find(x => x.id === payload.chatId);
        if (c) { c.messages.forEach(m => { if (!m.readBy.includes(payload.userId)) m.readBy.push(payload.userId); }); if (activeChatId === payload.chatId) renderMessages(); }
      } else if (event === 'chat:pin') {
        const c = chats.find(x => x.id === payload.chatId);
        if (c) { c.pinnedMessageId = payload.pinnedMessageId; if (activeChatId === payload.chatId) renderPinnedBar(c); }
      } else if (event === 'chat:new') {
        if (!chats.some(c => c.id === payload.id)) chats.unshift(payload);
        renderChats();
      } else if (event === 'typing' && activeChatId === payload.chatId) {
        const c = chats.find(x => x.id === payload.chatId);
        if (c) $('r-sub').textContent = payload.isTyping ? 'печатает...' : (c.type === 'dm' ? fmtLast(c.peer) : `${c.members.length} участн.`);
      } else if (event === 'presence') {
        chats.forEach(c => { if (c.peer?.id === payload.userId) { c.peer.online = payload.online; c.peer.lastSeen = payload.lastSeen; } });
        contacts.forEach(u => { if (u.id === payload.userId) { u.online = payload.online; u.lastSeen = payload.lastSeen; } });
        renderContacts();
      } else if (event === 'user:update') {
        currentUser = payload;
        applySettings();
        refreshAllData();
      } else if (event === 'call:signal') {
        onWsCall(payload);
      }
    } catch {}
  };
}

async function bootApp() {
  $('auth-screen').classList.add('hidden');
  $('app-shell').classList.remove('hidden');
  $('chats-skeleton').classList.remove('hidden');
  connectWS();

async function refreshAllData() {
  const [meR, chR, coR, stR, catR] = await Promise.all([
    api('GET', '/api/auth/me'),
    api('GET', '/api/chats'),
    api('GET', '/api/contacts'),
    api('GET', '/api/stories'),
    api('GET', '/api/catalog')
  ]);
  currentUser = meR.user;
  chats = chR.chats;
  contacts = coR.contacts;
  blockedUsers = coR.blocked;
  stories = stR.stories;
  catalog = catR;
  const uR = await api('GET', `/api/users/${currentUser.id}`);
  myGifts = uR.user.gifts || [];
  applySettings();
  renderAll();
}

function applySettings() {
  if (!currentUser) return;
  const s = currentUser.settings || {};
  document.body.classList.toggle('light', s.theme === 'light');
  document.documentElement.style.setProperty('--msg-fs', (s.fontSize || 15) + 'px');
  $('theme-sub').textContent = `${s.theme === 'light' ? 'Светлая' : 'Тёмная'} • ${s.fontSize || 15}px`;
  $('notif-sub').textContent = s.notifications !== false ? 'Включены' : 'Выключены';
  $('dl-sub').textContent = `Автозагрузка: ${s.autoDownload !== false ? 'Вкл' : 'Выкл'}`;
}

function renderAll() {
  if (!currentUser) return;
  const fullName = `${currentUser.firstName || ''} ${currentUser.lastName || ''}`.trim() || 'Пользователь';
  const avaHtml = currentUser.avatar ? `<img src="${esc(currentUser.avatar)}" alt="">` : esc((currentUser.firstName?.[0] || 'C').toUpperCase());
  $('nav-ava').innerHTML = avaHtml;
  $('side-ava').innerHTML = avaHtml;
  $('side-name').textContent = fullName;
  $('side-stars').textContent = (currentUser.stars || 0).toLocaleString('ru-RU');
  $('side-coins').textContent = (currentUser.ton || 0).toLocaleString('ru-RU');
  $('adminNav').classList.toggle('hidden', !currentUser.isAdmin);

  const tot = chats.filter(c => !c.archived).reduce((s, c) => s + (c.unread || 0), 0);
  $('nav-unread-badge').textContent = tot;
  $('nav-unread-badge').classList.toggle('hidden', tot === 0);
  $('side-unread-badge').textContent = tot;
  $('side-unread-badge').classList.toggle('hidden', tot === 0);

  renderContacts();
  renderStarsView();
  renderCatalogView();

}

function switchTab(t) {
  if (t === 'admin' && !currentUser?.isAdmin) {
    showToast('Доступ запрещён');
    return;
  }
  activeTab = t;
  ['chats', 'contacts', 'stars', 'gifts', 'ccoins', 'profile', 'settings', 'admin'].forEach(x => {
    $('view-' + x)?.classList.toggle('hidden', x !== t);
  });
  document.querySelectorAll('#sideNav [data-tab], .sidebar [data-tab]').forEach(btn => {
    btn.classList.toggle('active', btn.dataset.tab === t);
  });
  ['chats', 'contacts', 'gifts', 'profile', 'settings'].forEach(x => {
    $('nav-' + x)?.classList.toggle('active', x === t);
  });
  if (tabMeta[t]) {
    $('topbar-title').textContent = tabMeta[t][0];
    $('topbar-sub').textContent = tabMeta[t][1];
  }
  if (t === 'admin' && currentUser?.isAdmin) loadAdminOverview();
  window.scrollTo(0, 0);
}

async function publishTextStory() {
  const text = $('st-txt').value.trim();

    showToast('История опубликована');
  } catch (e) { showToast(e.message); }

function publishMediaStory(inp) {
  const f = inp.files[0]; if (!f) return;
  const r = new FileReader();
  r.onload = async e => {
    try {
      const up = await api('POST', '/api/upload', { dataUrl: e.target.result, fileName: f.name });
      await api('POST', '/api/stories', {
        type: f.type.startsWith('video') ? 'video' : 'photo',
        text: $('st-txt').value.trim(),
        mediaUrl: up.url,
        privacy: $('st-priv').value,
        savedToProfile: $('st-save').checked
      });
      closeModal();
      await refreshAllData();
      showToast('История опубликована');
    } catch (err) { showToast(err.message); }
  };
  r.readAsDataURL(f);
}

  $('chats-list').innerHTML = list.length ? list.map(c => {
    const last = c.messages[c.messages.length - 1];
    const sub = last
      ? (last.type === 'gift' ? `Подарок: ${last.giftSnapshot?.name || ''}` : last.type === 'photo' ? 'Фото' : last.type === 'video' ? 'Видео' : last.type === 'voice' ? 'Голосовое сообщение' : last.type === 'file' ? `Файл: ${last.fileName || ''}` : last.text)
      : 'Нет сообщений';

          <div class="c-r1">
            <div class="c-title">${esc(c.title)} ${c.peer?.verified ? verifiedSvg : ''} ${c.muted ? '<span class="muted small">(без звука)</span>' : ''}</div>

function renderMessages() {
  const c = chats.find(x => x.id === activeChatId); if (!c) return;
  const q = ($('r-search-inp').value || '').toLowerCase();
  const list = q ? c.messages.filter(m => (m.text || '').toLowerCase().includes(q)) : c.messages;
  const wrap = $('msgs-list');
  wrap.innerHTML = list.map(m => {
    if (m.type === 'system') return `<div class="msg sys">${esc(m.text)}</div>`;
    const isOut = m.senderId === currentUser.id;
    const rep = m.replyToId ? c.messages.find(x => x.id === m.replyToId) : null;
    const reacts = Object.entries(m.reactions || {}).map(([em, ids]) =>
      `<span class="r-tag" onclick="event.stopPropagation();reactMsg('${esc(m.id)}','${esc(em)}')">${esc(em)} ${ids.length}</span>`
    ).join('');
    let b = '';
    if (m.forwardedFrom) b += `<div class="muted small" style="margin-bottom:3px">Переслано от ${esc(m.forwardedFrom)}</div>`;
    if (rep) b += `<div class="r-box">${esc(rep.text || 'Медиа')}</div>`;
    if (m.type === 'photo' && m.mediaUrl) b += `<img class="m-img" src="${esc(m.mediaUrl)}" alt="">`;
    if (m.type === 'video' && m.mediaUrl) b += `<video class="m-video" src="${esc(m.mediaUrl)}" controls></video>`;
    if (m.type === 'voice' && m.mediaUrl) b += `<audio controls src="${esc(m.mediaUrl)}" style="max-width:220px;height:36px"></audio>`;
    if (m.type === 'file' && m.mediaUrl) b += `<a href="${esc(m.mediaUrl)}" download="${esc(m.fileName || 'file')}" style="color:var(--accent-soft);font-weight:600;display:block;margin-bottom:4px">Файл: ${esc(m.fileName || 'Файл')}</a>`;
    if (m.type === 'gift' && m.giftSnapshot) {
      const g = m.giftSna
